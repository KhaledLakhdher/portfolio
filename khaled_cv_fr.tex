\section{Projets}

\proj{Nexus AI - Plateforme de Gestion d'Employés en Intelligence Artificielle}
\begin{itemize}
  \item Plateforme multi-tenant où des agents LLM sont gérés comme des employés : permissions IAM, mémoire
        vectorielle hiérarchique, orchestration multi-agents avec supervision temps réel par WebSocket, et
        passerelle de modèles agnostique (Anthropic, OpenRouter, Gemini) exécutant les agents pour une
        fraction de centime par run. Stack : FastAPI, PostgreSQL/pgvector, Redis, Next.js.
\end{itemize}

\proj{DecisionFlow - Analyste Métier IA (SaaS Multi-Tenant)}
\begin{itemize}
  \item Transformation d'un CSV importé en entrepôt de données : nettoyage automatisé, schéma en étoile,
        prévisions et questions-réponses en langage naturel, sur un SaaS multi-tenant (FastAPI,
        PostgreSQL/RLS, DuckDB, Polars, Next.js) avec SQL généré par LLM isolé sur 3 couches et isolation
        des tenants appliquée en base de données.
\end{itemize}

\proj{CodeReview Evaluation - Réviseur de Code LLM Benchmarké}
\begin{itemize}
  \item Développement d'un framework de revue de code et d'évaluation LLM sur BugsInPy (501 bugs réels) ;
        sur 49 bugs, un LLM-as-judge montre qu'un score de détection unique surestime la capacité
        d'environ 1,8$\times$ (78\,\% de localisation contre 43\,\% de diagnostic), comparaison sur 3 niveaux de
        modèles, avec une démo Streamlit en direct.
\end{itemize}
